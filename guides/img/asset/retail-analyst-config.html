retail‑postgres
SAM SQL Connector
TCP / TLS · port 5432
retail database
PostgreSQL · AWS EC2 · us-east-2
connectors/retail-postgres.yaml 
kind: connector
name: retail-postgres
description: PostgreSQL connector for
  the retail database on EC2
spec:
  type: sql
  subtype: postgres
  values:
    database: retail
    hostname: ec2-3-129-17-126
      .us-east-2.compute
      .amazonaws.com
    port: 5432
    username: postgres
    password: ${RETAIL_DB_PASSWORD}
agents/retail-analyst.yaml 
kind: agent
name: retail-analyst
spec:
  systemPrompt: |
    You are a retail data
    analyst with SQL access
    to the retail database…
  connectors:
    - retail-postgres
  inputModes:  [text]
  outputModes: [text]
  deploy: true
